    • Percent rates are in percent units (e.g., 2.5 means 2.5%), unless a function explicitly states otherwise. """ from future import annotations

    def rmv_percent(df: pd.DataFrame, cols: Sequence[str]) -> pd.DataFrame: """Remove percent-signs from specified columns and convert to float.

    Parameters
----------
df : DataFrame
    Input frame that contains percentage-like text columns.
cols : sequence of str
    Column names to clean.

Returns
-------
DataFrame
    A copy of ``df[cols]`` with ``"%"`` removed (via regex) and dtype float.

Notes
-----
- Also trims whitespace and thousands separators (",") if present.
- Non-convertible values become ``NaN``.
"""
out = (
    df.loc[:, cols]
    .replace({"%": "", ",": ""}, regex=True)
    .apply(lambda s: pd.to_numeric(s, errors="coerce"))
)
return out

    def check(df: pd.DataFrame, start: Optional[str] = "2024Q1", end: Optional[str] = "2026Q2") -> pd.DataFrame: """Quickly slice a PeriodIndex-ed frame to inspect a join window.

    Useful for visual checks where the historical portion is in levels and the
forecast portion is in YoY percent change.

Parameters
----------
df : DataFrame
    Quarterly PeriodIndex-ed frame.
start, end : str or None
    Inclusive slice bounds like ``"2025Q1"``. Use ``None`` to omit a bound.

Returns
-------
DataFrame
    Sliced view.
"""
if start and end:
    return df.loc[start:end]
if start and not end:
    return df.loc[start:]
if end and not start:
    return df.loc[:end]
return df

    def get_page( pagename: str, filter_: str = "*", instance: str = "existing", show: bool | str = False, quarterly_period: bool = True, verbose: bool = False, ) -> pd.DataFrame: """Pull a page from the currently open EViews workfile.

    Parameters
----------
pagename : str
    EViews page name to pull.
filter_ : str, default "*"
    Wildcard filter for variable names.
instance : {"existing", "new"}
    See ``py2eviews.GetEViewsApp``.
show : bool | str, default False
    Show the EViews GUI window. Accepts truthy strings for backward
    compatibility with older code.
quarterly_period : bool, default True
    If True, convert the DatetimeIndex to a quarterly PeriodIndex.
verbose : bool, default False
    If True, print a tail preview.

Returns
-------
DataFrame
    Data fetched from EViews, optionally reindexed to ``PeriodIndex(freq='Q')``.
"""
if evp is None:  # pragma: no cover
    raise ImportError("py2eviews is not installed/available in this environment.")

app = evp.GetEViewsApp(instance=instance, showwindow=bool(show))
df = evp.GetWFAsPython(app=app, pagename=pagename, namefilter=filter_)
if quarterly_period:
    df.index = pd.PeriodIndex(df.index, freq="Q")
if verbose:
    print(df.tail())
return df

    def select_page(page_name: str, instance: str = "existing", show: bool | str = False) -> None: """Select an EViews page in the current workfile.

    Parameters
----------
page_name : str
    Page to select.
instance, show : see :func:`get_page`.
"""
if evp is None:  # pragma: no cover
    raise ImportError("py2eviews is not installed/available in this environment.")

app = evp.GetEViewsApp(instance=instance, showwindow=bool(show))
cmd = f"pageselect {page_name}"
evp.Run(cmd, app=app)

    def push(df: pd.DataFrame, instance: str = "existing", show: bool | str = False, newwf: bool = False) -> None: """Push a DataFrame to the current EViews workfile page.

    Index handling
--------------
- If the frame is indexed by quarterly Periods, convert to their start-of-
  quarter timestamps (EViews-friendly) before pushing.

Parameters
----------
df : DataFrame
    Data to push.
instance, show : see :func:`get_page`.
newwf : bool, default False
    If True, create a new workfile; otherwise push to current page.
"""
if evp is None:  # pragma: no cover
    raise ImportError("py2eviews is not installed/available in this environment.")

app = evp.GetEViewsApp(instance=instance, showwindow=bool(show))
out = df.copy()
if isinstance(out.index, pd.PeriodIndex) and out.index.freqstr.upper().startswith("Q"):
    out.index = out.index.to_timestamp(how="start")
evp.PutPythonAsWF(out, app=app, newwf=newwf)

    def get_index(df: pd.DataFrame, date: str | pd.Period) -> int: """Return the integer location of date in the index.

    def rename_rbind( df1: pd.DataFrame, df2: pd.DataFrame, first_forecast_date: str, dropna: bool = False, verbose: bool = False, ) -> pd.DataFrame: """Row-bind two frames across a forecast join, aligning column names.

    ``df1`` is treated as the historical/TD baseline; ``df2`` contains a
forecast that starts at ``first_forecast_date`` (inclusive).

Parameters
----------
df1, df2 : DataFrame
    Must have the same number of columns.
first_forecast_date : str
    Quarter label, e.g. ``"2025Q4"``.
dropna : bool, default False
    Drop any rows containing NA after concatenation.
verbose : bool, default False
    Print helpful context.

Returns
-------
DataFrame
"""
period = pd.Period(first_forecast_date, freq="Q")

if df1.shape[1] != df2.shape[1]:
    raise ValueError("DataFrames must have the same number of columns.")

df1_hist = df1.loc[: period - 1]
df2_fcst = df2.loc[period:].copy()
df2_fcst.columns = df1_hist.columns

out = pd.concat([df1_hist, df2_fcst], axis=0)
if dropna:
    out = out.dropna(how="any")
if verbose:
    print("Joined at", period)
return out

    def annualize(df: pd.DataFrame, col: str, join_to_df: bool = True) -> pd.DataFrame | pd.Series: """Compute annualized growth from a level column using pct_change.

    Formula: ``((1 + qoq) ** 4 - 1) * 100`` where ``qoq = df[col].pct_change()``.

Parameters
----------
df : DataFrame
col : str
    Column that represents **levels**.
join_to_df : bool, default True
    If True, add a new column ``f"{col}_ANN"`` and return the frame;
    else return the computed Series.
"""
ann = ((1 + df[col].pct_change()) ** 4 - 1) * 100
if join_to_df:
    df = df.copy()
    df[f"{col}_ANN"] = ann
    return df
return ann

    def join(df1: pd.DataFrame | pd.Series, df2: pd.DataFrame | pd.Series, *, bind: str, use_name: int = 1) -> pd.DataFrame: """Concatenate two objects either by rows (bind='r') or columns (bind='c').

    Parameters
----------
df1, df2 : Series or DataFrame
bind : {'r', 'c'}
    ``'r'`` → row-bind (stack); ``'c'`` → column-bind (side-by-side).
use_name : {1, 2}, default 1
    For row-binding with mismatched column names, choose whose column names
    to keep: 1 → use ``df1`` names; 2 → use ``df2`` names.

Returns
-------
DataFrame
"""
d1 = df1.to_frame() if isinstance(df1, pd.Series) else df1.copy()
d2 = df2.to_frame() if isinstance(df2, pd.Series) else df2.copy()

if bind == "r":
    if d1.shape[1] != d2.shape[1]:
        raise ValueError("Row-bind requires same number of columns.")
    if use_name == 1:
        d2.columns = d1.columns
    elif use_name == 2:
        d1.columns = d2.columns
    axis = 0
elif bind == "c":
    axis = 1
else:
    raise ValueError("bind must be 'r' or 'c'.")

return pd.concat([d1, d2], axis=axis)

    def yoy( df: pd.DataFrame | pd.Series, *, label: str = "_yoy", periods: int = 4, verbose: bool = False, ) -> pd.DataFrame: """Compute year-on-year % change using log-differences (×100).

    Parameters
----------
df : Series or DataFrame of **levels**
label : str, default "_yoy"
    Suffix to add to resulting column names (DataFrame case).
periods : int, default 4
    Number of periods to lag for YoY (4 for quarterly data).
verbose : bool, default False
    Print a preview.

Returns
-------
DataFrame
    If input is a Series, returns a single-column DataFrame for consistency.
"""
x = df.to_frame() if isinstance(df, pd.Series) else df
temp = (np.log(x) - np.log(x).shift(periods)) * 100
if label:
    temp.columns = [c if c.endswith(label) else f"{c}{label}" for c in temp.columns]
if verbose:
    print(temp.dropna().tail())
return temp

    def to_q(df: pd.DataFrame) -> pd.DataFrame: """Ensure the index is a quarterly PeriodIndex.

    Returns a **copy** with ``freq='Q'`` PeriodIndex.
"""
out = df.copy()
out.index = pd.PeriodIndex(out.index, freq="Q")
return out

    def suffix(df: pd.DataFrame, label: str = "_yoy") -> pd.DataFrame: """Append label to column names if not already present.""" out = df.copy() out.columns = [c if c.endswith(label) else f"{c}{label}" for c in out.columns] return out

    def conv_date(df: pd.DataFrame, period: str = "Q") -> pd.DataFrame: """Convert a DatetimeIndex to a PeriodIndex.

    Currently supports quarterly (``period='Q'``).
"""
if period.upper() != "Q":
    raise NotImplementedError("Only quarterly conversion is supported.")
out = df.copy()
out.index = pd.PeriodIndex(pd.to_datetime(out.index), freq="Q")
return out

    def repeat_last_row(df: pd.DataFrame, last_idx: str | pd.Period, n: int = 14) -> pd.DataFrame: """Extend a frame by repeating the last row n more quarters.

    Parameters
----------
df : DataFrame
last_idx : str or Period
    Label of the last historical quarter to anchor on.
n : int, default 14
    Number of quarters to append by forward-filling the last row.
"""
out = df.copy()
last_idx = pd.Period(last_idx, freq="Q") if isinstance(last_idx, str) else last_idx
start = out.index[0]
extended_index = pd.period_range(start=start, periods=len(out) + n, freq="Q")
out = out.reindex(extended_index)
out.loc[last_idx:] = out.loc[last_idx:].ffill()
return out

    def demean(df: pd.DataFrame) -> pd.DataFrame: """Column-wise de-meaned copy of df.""" return df - df.mean()

    def yoy_to_levels( yoy_rates_pct: pd.DataFrame, , first_forecast_date: str, levels_start: pd.DataFrame, annual: bool = True, lag: int = 4, verbose: bool = False, ) -> pd.DataFrame: """Convert YoY or QoQ growth paths into levels* after a join point.

    Parameters
----------
yoy_rates_pct : DataFrame
    Growth rates in percent units. If ``annual=True``, treat values as YoY
    growth; otherwise as QoQ growth.
first_forecast_date : str
    The first quarter **to be computed** using the growth rates (e.g.,
    ``"2025Q4"``). ``levels_start`` must include the prior quarter level.
levels_start : DataFrame
    Historical **levels** through at least the quarter **before**
    ``first_forecast_date`` for the same columns as ``yoy_rates_pct``.
annual : bool, default True
    Interpret input rates as YoY if True; QoQ if False.
lag : int, default 4
    Number of quarters in a year, used if ``annual=False``.
verbose : bool, default False
    Print a tail preview.

Returns
-------
DataFrame
    Levels series combining ``levels_start`` (up to join-1) and simulated
    levels thereafter.

Notes
-----
- For annual YoY rates ``r_t`` (in percent), quarterly growth is
  ``g_t = (1 + r_t/100)**(1/4) - 1`` and ``L_t = L_{t-1} * (1 + g_t)``.
- For QoQ rates ``q_t`` (percent), ``L_t = L_{t-1} * (1 + q_t/100)``.
"""
period = pd.Period(first_forecast_date, freq="Q")

# Align and prepare containers
levels = levels_start.copy()
rates = yoy_rates_pct.copy()

# Ensure indexes are PeriodIndex(Q)
levels.index = pd.PeriodIndex(levels.index, freq="Q")
rates.index = pd.PeriodIndex(rates.index, freq="Q")

# We will iteratively fill from period onward
all_index = levels.index.union(rates.index)
levels = levels.reindex(all_index)

# Iterate only over the forecast path for clarity and stability
cols = levels.columns.intersection(rates.columns)
if cols.empty:
    raise ValueError("No overlapping columns between levels_start and yoy_rates_pct.")

prev = period - 1
if prev not in levels.index:
    raise KeyError(
        f"levels_start must include the quarter prior to {period} (i.e., {prev})."
    )

for t in all_index[all_index.get_loc(period) :]:
    if t not in rates.index:
        # no new rate information; carry forward last level
        levels.loc[t, cols] = levels.loc[t - 1, cols]
        continue
    if annual:
        g = (1.0 + rates.loc[t, cols] / 100.0) ** (1.0 / 4.0) - 1.0
    else:
        g = rates.loc[t, cols] / 100.0
    levels.loc[t, cols] = levels.loc[t - 1, cols] * (1.0 + g)

out = levels.loc[: all_index[-1], cols]
if verbose:
    print(out.tail())
return out

    def annualize_qoq(qoq_pct: pd.Series | pd.DataFrame) -> pd.Series | pd.DataFrame: """Convert QoQ percent changes into annualized percent changes.

    Formula: ``(1 + qoq/100)**4 - 1`` → ×100 percent units.
"""
return ((1 + qoq_pct / 100.0) ** 4 - 1) * 100.0

    def read_mst_2025(path: str) -> pd.DataFrame: """Read a specific 2025 MST Excel package format into a tidy quarterly frame.

    Parameters
----------
path : str
    Path to the Excel file (sheet name "Fiscal Variable").

Returns
-------
DataFrame
    YoY percent changes with ``PeriodIndex(freq='Q')`` starting 2025Q4.
"""
ex_ = pd.read_excel(path, sheet_name="Fiscal Variable", header=None)
ex = ex_.T
ex.columns = ex.iloc[0]
ex = ex.drop(ex.index[0]).reset_index(drop=True)
mst_vars_yoy = ex.iloc[4:, 0:115]
dates = pd.period_range("2025Q4", periods=len(mst_vars_yoy), freq="Q")
mst_vars_yoy.index = dates
return mst_vars_yoy

    def read(path: str, index_col: Optional[str] = None, sheet_name: Optional[str] = None) -> pd.DataFrame: """Read CSV or Excel and return a quarterly PeriodIndex-ed frame.

    Parameters
----------
path : str
    File path. CSV is assumed when ``sheet_name`` is None.
index_col : str, optional
    Column to use as index for CSV files. Defaults to ``"_date_"``.
sheet_name : str, optional
    If provided, read the specified Excel sheet instead of CSV.
"""
if sheet_name is None:
    idx = "_date_" if index_col is None else index_col
    df = pd.read_csv(path, index_col=idx)
    df.index = pd.to_datetime(df.index).to_period("Q")
    return df
else:
    return pd.read_excel(path, sheet_name=sheet_name)

    def read_mst(path: str) -> pd.DataFrame: """Read an MST CSV (OSFI stressed data) and return CA/US YoY percent changes.

    Expected CSV layout is transposed so that the useful variables are in rows
after transformation.
"""
df_mst = pd.read_csv(path).T
df_mst.columns = ["a", "CA", "US", "b", "c"]
df_mst = df_mst[["CA", "US"]].iloc[5:25]
q_idx = pd.period_range(start="2025Q4", end="2030Q3", freq="Q")
df_mst.index = q_idx
mst_given_yoy = rmv_percent(df_mst, ["CA", "US"]).rename(
    columns={"CA": "CPICA_MST", "US": "CPIUS_MST"}
)
return mst_given_yoy

    -----------------------------------------------------------------------------

    Donor–target stress propagation (simple linear beta model)

    -----------------------------------------------------------------------------

    def ols_betas(y: np.ndarray, X: np.ndarray, *, add_intercept: bool = False) -> np.ndarray: """Estimate OLS coefficients for y ~ X using least squares.

    Parameters
----------
y : array-like, shape (T,)
X : array-like, shape (T, K)
add_intercept : bool, default False
    If True, include a column of ones in ``X``.
"""
if y.ndim != 1:
    y = y.squeeze()
if add_intercept:
    X = np.column_stack([np.ones(len(X)), X])
beta, *_ = np.linalg.lstsq(X, y, rcond=None)
return beta

    def main(hist: pd.DataFrame, scen: pd.DataFrame) -> pd.DataFrame: """Estimate donor betas on history and apply them to scenario shocks.

    Targets: UK, AU, EA, JP; Donors: US, CA. All series are YoY percent changes.

Parameters
----------
hist : DataFrame
    Historical YoY percent-change data with columns:
    ``['CPIUS_TDBS','CPICA_TDBS','CPIUK_TDBS','CPIAU_TDBS','CPIEA_TDBS','CPIJP_TDBS']``
scen : DataFrame
    Scenario window containing baseline (TDBS) and stressed (MST) YoY for US
    and CA, and baseline for targets.

Returns
-------
DataFrame
    Stressed YoY for targets: ``CPIUK_MST, CPIAU_MST, CPIEA_MST, CPIJP_MST``
    with a quarterly PeriodIndex.
"""
required_hist_cols = [
    "CPIUS_TDBS",
    "CPICA_TDBS",
    "CPIUK_TDBS",
    "CPIAU_TDBS",
    "CPIEA_TDBS",
    "CPIJP_TDBS",
]
missing = [c for c in required_hist_cols if c not in hist.columns]
if missing:
    raise ValueError(f"history is missing columns: {missing}")

Z = demean(hist[required_hist_cols].dropna())
donors = Z[["CPIUS_TDBS", "CPICA_TDBS"]].to_numpy()

beta = {
    "UK": ols_betas(Z["CPIUK_TDBS"].to_numpy(), donors),
    "AU": ols_betas(Z["CPIAU_TDBS"].to_numpy(), donors),
    "EA": ols_betas(Z["CPIEA_TDBS"].to_numpy(), donors),
    "JP": ols_betas(Z["CPIJP_TDBS"].to_numpy(), donors),
}

required_scen_cols = [
    "CPICA_TDBS",
    "CPIUS_TDBS",
    "CPIUK_TDBS",
    "CPIAU_TDBS",
    "CPIEA_TDBS",
    "CPIJP_TDBS",
    "CPIUS_MST",
    "CPICA_MST",
]
missing = [c for c in required_scen_cols if c not in scen.columns]
if missing:
    raise ValueError(f"scenario is missing columns: {missing}")

scen = scen.copy()
scen["US_SHOCK"] = scen["CPIUS_MST"] - scen["CPIUS_TDBS"]
scen["CA_SHOCK"] = scen["CPICA_MST"] - scen["CPICA_TDBS"]

shocks = scen[["US_SHOCK", "CA_SHOCK"]].to_numpy()

scen["CPIUK_MST"] = scen["CPIUK_TDBS"].to_numpy() + shocks @ beta["UK"]
scen["CPIAU_MST"] = scen["CPIAU_TDBS"].to_numpy() + shocks @ beta["AU"]
scen["CPIEA_MST"] = scen["CPIEA_TDBS"].to_numpy() + shocks @ beta["EA"]
scen["CPIJP_MST"] = scen["CPIJP_TDBS"].to_numpy() + shocks @ beta["JP"]

out_cols = ["CPIUS_MST", "CPICA_MST", "CPIUK_MST", "CPIAU_MST", "CPIEA_MST", "CPIJP_MST"]
smc_mst_yoy = scen[out_cols].copy()
smc_mst_yoy.index = pd.PeriodIndex(smc_mst_yoy.index, freq="Q")

print(
    "Estimated betas:\n"
    f" UK on US, CA: US={beta['UK'][0]:.3f}, CA={beta['UK'][1]:.3f}\n"
    f" AU on US, CA: US={beta['AU'][0]:.3f}, CA={beta['AU'][1]:.3f}\n"
    f" EA on US, CA: US={beta['EA'][0]:.3f}, CA={beta['EA'][1]:.3f}\n"
    f" JP on US, CA: US={beta['JP'][0]:.3f}, CA={beta['JP'][1]:.3f}"
)
return smc_mst_yoy

    -----------------------------------------------------------------------------

    End-to-end MST levels builder (example wiring)

    -----------------------------------------------------------------------------

    def mst_levels( pull_page: str, mst_forecast_yoy: pd.DataFrame, mst_vars: Sequence[str], jump_off: str, n: int = 13, ) -> pd.DataFrame: """Construct MST levels series by combining actuals + stressed YoY.

    Steps (high-level)
------------------
1. Pull baseline (TDBS) levels from EViews page ``pull_page``.
2. Compute YoY for actuals and baseline forecast; splice in MST YoY from
   ``mst_forecast_yoy`` for the selected variables ``mst_vars``.
3. Estimate donor betas on history and propagate US/CA shocks to targets.
4. Extend to 2030Q3 + ``n`` quarters by repeating last row.
5. Convert YoY to levels using :func:`yoy_to_levels` from ``jump_off``.

Parameters
----------
pull_page : str
    EViews page name to pull.
mst_forecast_yoy : DataFrame
    Stressed YoY in percent with columns like ``<VAR>_MST``.
mst_vars : sequence of str
    Variable base names (e.g., ``["CPIAU","CPICA",...]``).
jump_off : str
    Quarter where the MST forecast starts (e.g., ``"2025Q4"``).
n : int, default 13
    Number of quarters to extend by repeating the last row.
"""
jump_off_p = pd.Period(jump_off, freq="Q")

# Build lists of baseline/stress names
given_tdbs = [f"{v}_TDBS" for v in mst_vars]
given_mst = [f"{v}_MST" for v in mst_vars]

# 1) Pull baseline levels
tdbs = get_page(pull_page)

# 2) Compute YoY
actuals = tdbs.loc[: jump_off_p - 1]
tdbs_yoy = yoy(tdbs, label="")

# Align actuals YoY to stress names and splice MST forecast
actuals_yoy = yoy(actuals[given_tdbs], label="")
actuals_yoy.columns = given_mst
mst_actuals_yoy = join(actuals_yoy, mst_forecast_yoy[given_mst], bind="r")

scenario_yoy = join(tdbs_yoy, mst_actuals_yoy, bind="c")
hist = tdbs_yoy
scen = scenario_yoy.loc[jump_off_p:].copy()

# 3) Donor propagation
smc_mst_yoy = main(hist, scen)

# 4) Build levels container up to 2030Q3 and extend
